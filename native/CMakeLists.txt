cmake_minimum_required(VERSION 3.20)
project(visualizeit_pose VERSION 0.1.0 LANGUAGES CXX)
add_library(visualizeit_pose SHARED src/pose.cpp)
target_include_directories(visualizeit_pose PUBLIC include)
target_compile_features(visualizeit_pose PUBLIC cxx_std_17)
target_compile_definitions(visualizeit_pose PRIVATE VI_SHARED)
include(CTest)
if(BUILD_TESTING AND NOT ANDROID AND NOT IOS)
  add_executable(pose_tests tests/pose_tests.cpp)
  target_link_libraries(pose_tests PRIVATE visualizeit_pose)
  add_test(NAME pose_coordinates COMMAND pose_tests)
endif()
